# IAM_Base_Section

This is for reference when creating a new section file, the base classes should normally not have to be changed.

```php
// Exit if accessed directly
if (!defined('ABSPATH')) {
    exit;
}

abstract class IAM_Base_Section
{
    /* Each section needs these variables, the use of these variables
     * is shown in comments in the other docs.
     */
    protected $section_id;
    protected $section_title;
    protected $section_class;
    protected $fields = [];

    /* The instantiation is explained in 6. IAM_Base_Page */
    private static $instances = [];
    public static function instance()
    {
      $calledClass = static::class;
        if (!isset(self::$instances[$calledClass])) {
          self::$instances[$calledClass] = new static();
        }
        return self::$instances[$calledClass];
    }

    /* The construct is explained in 6. IAM_Base_Page and follows similar logic */
    public function __construct($section_id, $section_title, $section_class)
    {
        $this->section_id = $section_id;
        $this->section_title = $section_title;
        $this->section_class = $section_class;
    }

    /* This is the function that gets used in `IAM_Base_Page::register_settings()`
     * For each section the add_settings_section() function gets called.
     *
     * Then it loops through the fields added from the section file into the $fields array.
     *
     * https://developer.wordpress.org/reference/functions/add_settings_section/
     * https://developer.wordpress.org/reference/functions/add_settings_field/
     */
    public function register_section($menu_slug)
    {
        /* https://developer.wordpress.org/reference/functions/add_settings_section/ */
        add_settings_section(
            /* $id */
            $this->section_id,
            /* $title */
            $this->section_title,
            /* $callback */
            [$this, 'section_callback'],
            /* $page */
            $menu_slug,
            /* $args['before_section', 'after_section'] */
            ['before_section' => '<div class="' . $this->section_class . '">', 'after_section' => '</div>']
        );

        /* https://developer.wordpress.org/reference/functions/add_settings_field/ */
        foreach ($this->fields as $field) {
            add_settings_field(
                /* $id */
                $field['id'],
                /* $title */
                $field['title'],
                /* $callback */
                $field['callback'],
                /* $page */
                $menu_slug,
                /* $section */
                $this->section_id,
                /* $args['label_for','class'] */
                $field['args']
            );

            /* https://developer.wordpress.org/reference/hooks/register_setting/ */
            register_setting($menu_slug . '_options_group', $field['id']);
        }
    }

    /* This is the function that gets used in a new section file to add items
     * to the $fields array so they can get loaded during `IAM_Base_Section::register_section()`
     */
    public function add_field($field_id, $field_title, $callback, $args = [])
    {
        $this->fields[] = [
            'id' => $field_id,
            'title' => $field_title,
            'callback' => $callback,
            'args' => $args,
        ];
    }

    /* This is the html output callback for what prepends the table that gets output by
     * `IAM_Base_page::render_page()` do_settings_sections().
     *
     * Like the `IAM_Base_Page::render_page()` function the `section_callback()` can be
     * overwritten inside a new section file.
     */
    public function section_callback()
    {
        echo '<p>Section description here.</p>';
    }
}
```
